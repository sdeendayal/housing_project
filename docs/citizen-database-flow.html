-- District SELECT DistrictName FROM districts WHERE DistrictId = 291; -- City SELECT CityName FROM cities WHERE CityId = 3252; -- Sector / Ward SELECT SectorName FROM sectors WHERE SectorId = 686; -- EM Office SELECT BranchName FROM em_offices WHERE BranchId = 3205;
-- Ek saath JOIN se sab naam SELECT pad.AssetId, d.DistrictName, c.CityName, s.SectorName, eo.BranchName FROM property_auction_detail pad LEFT JOIN districts d ON pad.DistrictId = d.DistrictId LEFT JOIN cities c ON pad.CityId = c.CityId LEFT JOIN sectors s ON pad.SectorId = s.SectorId LEFT JOIN em_offices eo ON pad.BranchId = eo.BranchId WHERE pad.AssetId = 147462;
Ward vs Sector: Alag ward table NAHI hai. Sirf sectors table hai. SectorName mein kabhi "Ward 6", kabhi "Sector 28", kabhi gaon ka naam (Malerna) hota hai. Ward hai ya Sector — SectorName padh kar pata chalta hai.
